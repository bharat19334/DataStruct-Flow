<h2 align="center">🏗️ SYSTEM ARCHITECTURE</h2>

```mermaid
flowchart TB
    U((Learner))

    subgraph FRONTEND["Frontend Layer"]
        UI["HTML / CSS / JavaScript"]
        AI_UI["AI Tutor"]
        VIS["Algorithm Visualizer"]
        PRACTICE["Practice Sheet"]
        IDE["Code Editor"]
        LS[("Local Storage")]
    end

    subgraph BACKEND["FastAPI Backend"]
        API["REST API"]
        CACHE["Response Cache"]
        HISTORY["History Service"]
        EXEC["Execution Router"]
    end

    GROQ["Groq AI"]
    TTS["ElevenLabs"]
    PY["Python Runtime"]
    PISTON["Piston API"]
    DB[("Supabase")]

    U --> UI
    UI --> AI_UI
    UI --> VIS
    UI --> PRACTICE
    UI --> IDE

    AI_UI --> API
    VIS --> API
    IDE --> API

    API --> CACHE
    API --> GROQ
    API --> TTS
    API --> EXEC

    EXEC --> PY
    EXEC --> PISTON

    API --> HISTORY
    HISTORY --> DB

    UI <--> LS
